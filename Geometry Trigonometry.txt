\begin{enumerate}[label=\textbf{(\Alph*)}]
    \item $(x - 3)^2 + (y + 4)^2 = 49$
    \item $(x + 3)^2 + (y - 4)^2 = 49$
    \item $(x - 3)^2 + (y + 4)^2 = 7$
    \item $(x + 3)^2 + (y - 4)^2 = 7$
\end{enumerate}
\end{multicols}
